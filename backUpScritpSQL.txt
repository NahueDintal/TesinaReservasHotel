-- ==========================================
-- BASE DE DATOS - HOTEL
-- ==========================================

DROP DATABASE IF EXISTS HotelDataBase;

CREATE DATABASE HotelDataBase;
USE HotelDataBase;


-- ==========================================
-- TABLAS INDEPENDIENTES
-- ==========================================

-- ==========================================
-- Tabla: DocumentType
-- ==========================================

CREATE TABLE DocumentType (
    idDocumentType INT AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    PRIMARY KEY (idDocumentType)
);


-- ==========================================
-- Tabla: Country
-- ==========================================

CREATE TABLE Country (
    idCountry INT AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    PRIMARY KEY (idCountry)
);


-- ==========================================
-- Tabla: CustomerStatus
-- ==========================================

CREATE TABLE CustomerStatus (
    idCustomerStatus INT AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    PRIMARY KEY (idCustomerStatus)
);


-- ==========================================
-- Tabla: CustomerOrigin
-- ==========================================

CREATE TABLE CustomerOrigin (
    idCustomerOrigin INT AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    PRIMARY KEY (idCustomerOrigin)
);


-- ==========================================
-- Tabla: ReservationStatus
-- ==========================================

CREATE TABLE ReservationStatus (
    idReservationStatus INT NOT NULL AUTO_INCREMENT,
    name VARCHAR(30) NOT NULL,
    PRIMARY KEY (idReservationStatus)
);


-- ==========================================
-- Tabla: ReservationType
-- ==========================================

CREATE TABLE ReservationType (
    idReservationType INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL
);


-- ==========================================
-- Tabla: PaymentMethod
-- ==========================================

CREATE TABLE PaymentMethod (
    idPaymentMethod INT NOT NULL AUTO_INCREMENT,
    name VARCHAR(50) NOT NULL,
    PRIMARY KEY (idPaymentMethod)
);


-- ==========================================
-- Tabla: PaymentStatus
-- ==========================================

CREATE TABLE PaymentStatus (
    idPaymentStatus INT NOT NULL AUTO_INCREMENT,
    name VARCHAR(50) NOT NULL,
    PRIMARY KEY (idPaymentStatus)
);


-- ==========================================
-- Tabla: Service
-- ==========================================

CREATE TABLE Service (
    idService INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    description VARCHAR(255),
    price DECIMAL(10,2) NOT NULL,
    active BOOLEAN NOT NULL DEFAULT TRUE
);


-- ==========================================
-- Tabla: Product
-- ==========================================

CREATE TABLE Product (
    idProduct INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    description VARCHAR(255),
    price DECIMAL(10,2) NOT NULL,
    active BOOLEAN NOT NULL DEFAULT TRUE
);


-- ==========================================
-- Tabla: ConsumptionType
-- ==========================================

CREATE TABLE ConsumptionType (
    idConsumptionType INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL
);


-- ==========================================
-- Tabla: ConsumptionStatus
-- ==========================================

CREATE TABLE ConsumptionStatus (
    idConsumptionStatus INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL
);


-- ==========================================
-- Tabla: JobPosition
-- ==========================================

CREATE TABLE JobPosition (
    idPosition INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL UNIQUE
);

-- ==========================================
-- Tabla: Department
-- ==========================================

CREATE TABLE Department (
    idDepartment INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL UNIQUE
);

-- ==========================================
-- Tabla: Shift  (NUEVA - requerida por StaffDAO)
-- ==========================================

CREATE TABLE Shift (
    idShift INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL UNIQUE
);

-- ==========================================
-- Tabla: RoomType
-- ==========================================


CREATE TABLE RoomType (
    idRoomType INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL UNIQUE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ==========================================
-- Tabla: RoomView
-- ==========================================

CREATE TABLE RoomView (
    idRoomView INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ==========================================
-- Tabla: Feature
-- ==========================================


CREATE TABLE Feature (
    idFeature INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ==========================================
-- DEPENDIENTES 1
-- ==========================================

-- ==========================================
-- Tabla: Customer
-- ==========================================

CREATE TABLE Customer (
    idCustomer INT AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    surname VARCHAR(100) NOT NULL,
    idDocumentType INT NOT NULL,
    documentNumber VARCHAR(30) NOT NULL,
    phoneNumber VARCHAR(30),
    email VARCHAR(255),
    idCountry INT NOT NULL,
    idCustomerStatus INT NOT NULL,
    idCustomerOrigin INT NOT NULL,
    PRIMARY KEY (idCustomer),
    FOREIGN KEY (idDocumentType) REFERENCES DocumentType(idDocumentType),
    FOREIGN KEY (idCountry) REFERENCES Country(idCountry),
    FOREIGN KEY (idCustomerStatus) REFERENCES CustomerStatus(idCustomerStatus),
    FOREIGN KEY (idCustomerOrigin) REFERENCES CustomerOrigin(idCustomerOrigin)
);


-- ==========================================
-- Tabla: Room (MODIFICADA con idRoom y active)
-- ==========================================

CREATE TABLE Room (
    idRoom INT AUTO_INCREMENT PRIMARY KEY,
    number INT NOT NULL,
    floor INT NOT NULL,
    idRoomType INT NOT NULL,
    capacity INT NOT NULL,
    idRoomView INT NOT NULL,
    available BOOLEAN NOT NULL DEFAULT TRUE,
    outOfService BOOLEAN NOT NULL DEFAULT FALSE,
    active BOOLEAN NOT NULL DEFAULT TRUE,
    price DECIMAL(10,2) NOT NULL,
    description TEXT,
    FOREIGN KEY (idRoomType) REFERENCES RoomType(idRoomType),
    FOREIGN KEY (idRoomView) REFERENCES RoomView(idRoomView)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ==========================================
-- Tabla: RoomFeature (MODIFICADA para usar idRoom)
-- ==========================================

CREATE TABLE RoomFeature (
    idRoom INT NOT NULL,
    idFeature INT NOT NULL,
    PRIMARY KEY (idRoom, idFeature),
    FOREIGN KEY (idRoom) REFERENCES Room(idRoom) ON DELETE CASCADE,
    FOREIGN KEY (idFeature) REFERENCES Feature(idFeature) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ==========================================
-- DEPENDIENTES 2
-- ==========================================

-- ==========================================
-- Tabla: Reservation
-- ==========================================

CREATE TABLE Reservation (
    idReservation INT AUTO_INCREMENT PRIMARY KEY,
    idCustomer INT NOT NULL,
    creationDate DATETIME NOT NULL,
    checkIn DATE NOT NULL,
    checkOut DATE NOT NULL,
    idReservationStatus INT NOT NULL,
    idReservationType INT NOT NULL,
    numberOfGuests INT NOT NULL,
    totalRate DECIMAL(10,2) NOT NULL,
    observations VARCHAR(500),
    CONSTRAINT fk_reservation_customer FOREIGN KEY (idCustomer) REFERENCES Customer(idCustomer),
    CONSTRAINT fk_reservation_status FOREIGN KEY (idReservationStatus) REFERENCES ReservationStatus(idReservationStatus),
    CONSTRAINT fk_reservation_type FOREIGN KEY (idReservationType) REFERENCES ReservationType(idReservationType)
);


-- ==========================================
-- Tabla: Staff
-- ==========================================

CREATE TABLE Staff (
    id VARCHAR(10) PRIMARY KEY,
    firstName VARCHAR(100) NOT NULL,
    lastName VARCHAR(100) NOT NULL,
    dni VARCHAR(20) NOT NULL UNIQUE,
    birthDate DATE,
    phone VARCHAR(20),
    email VARCHAR(150),
    street VARCHAR(150),
    addressNumber VARCHAR(20),
    city VARCHAR(100),
    idPosition INT NOT NULL,
    idDepartment INT NULL,
    idShift INT NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'ACTIVE',
    hireDate DATE,
    shiftStart TIME,
    shiftEnd TIME,
    salary DECIMAL(12,2),
    CONSTRAINT fk_staff_position   FOREIGN KEY (idPosition)   REFERENCES JobPosition(idPosition),
    CONSTRAINT fk_staff_department FOREIGN KEY (idDepartment) REFERENCES Department(idDepartment),
    CONSTRAINT fk_staff_shift      FOREIGN KEY (idShift)      REFERENCES Shift(idShift)
);

-- ==========================================
-- Tabla: StaffHistory  (NUEVA - requerida por StaffHistoryDAO)
-- ==========================================

CREATE TABLE StaffHistory (
    idHistory INT AUTO_INCREMENT PRIMARY KEY,
    idStaff VARCHAR(10) NOT NULL,
    changeType VARCHAR(50) NOT NULL,
    description VARCHAR(255),
    changedAt TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_staff_history_staff FOREIGN KEY (idStaff) REFERENCES Staff(id) ON DELETE CASCADE
);

-- ==========================================
-- DEPENDIENTES 3
-- ==========================================

-- ==========================================
-- Tabla: Payment
-- ==========================================

CREATE TABLE Payment (
    idPayment INT NOT NULL AUTO_INCREMENT,
    idReservation INT NOT NULL,
    amount DECIMAL(10,2) NOT NULL,
    paymentDate DATETIME NOT NULL,
    idPaymentMethod INT NOT NULL,
    idPaymentStatus INT NOT NULL,
    observations TEXT,
    PRIMARY KEY (idPayment),
    KEY idReservation (idReservation),
    KEY idPaymentMethod (idPaymentMethod),
    KEY idPaymentStatus (idPaymentStatus),
    CONSTRAINT payment_ibfk_1 FOREIGN KEY (idReservation) REFERENCES Reservation(idReservation),
    CONSTRAINT payment_ibfk_2 FOREIGN KEY (idPaymentMethod) REFERENCES PaymentMethod(idPaymentMethod),
    CONSTRAINT payment_ibfk_3 FOREIGN KEY (idPaymentStatus) REFERENCES PaymentStatus(idPaymentStatus)
);


-- ==========================================
-- Tabla: Consumption
-- ==========================================

CREATE TABLE Consumption (
    idConsumption INT AUTO_INCREMENT PRIMARY KEY,
    idReservation INT NOT NULL,
    idConsumptionType INT NOT NULL,
    idProduct INT NULL,
    idService INT NULL,
    quantity INT NOT NULL,
    unitPrice DECIMAL(10,2) NOT NULL,
    total DECIMAL(10,2) NOT NULL,
    consumptionDate DATETIME NOT NULL,
    idPaymentStatus INT NOT NULL,
    idConsumptionStatus INT NOT NULL DEFAULT 1,
    observations VARCHAR(500) NULL,
    CONSTRAINT fk_consumption_reservation FOREIGN KEY (idReservation) REFERENCES Reservation(idReservation),
    CONSTRAINT fk_consumption_type FOREIGN KEY (idConsumptionType) REFERENCES ConsumptionType(idConsumptionType),
    CONSTRAINT fk_consumption_product FOREIGN KEY (idProduct) REFERENCES Product(idProduct),
    CONSTRAINT fk_consumption_service FOREIGN KEY (idService) REFERENCES Service(idService),
    CONSTRAINT fk_consumption_payment_status FOREIGN KEY (idPaymentStatus) REFERENCES PaymentStatus(idPaymentStatus),
    CONSTRAINT fk_consumption_status FOREIGN KEY (idConsumptionStatus) REFERENCES ConsumptionStatus(idConsumptionStatus)
);
